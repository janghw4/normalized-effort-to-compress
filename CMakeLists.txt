cmake_minimum_required(VERSION 3.16)
project(normalized_etc VERSION 0.1.0 LANGUAGES C)
add_library(etc SHARED src/etc.c)
target_include_directories(etc PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/src)
set_target_properties(etc PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED YES)
if(MSVC)
  target_compile_options(etc PRIVATE /W4)
else()
  target_compile_options(etc PRIVATE -Wall -Wextra -Wpedantic)
endif()
